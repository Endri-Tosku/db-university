///////////// QUERRY CON SELECT ///////////

-- 1. Studenti nati nel 1990
SELECT *
FROM students
WHERE YEAR(date_of_birth) = 1990;

-- querry con count
SELECT COUNT(id)
FROM students
WHERE YEAR(date_of_birth) = 1990;



-- 2. Corsi con più di 10 crediti
SELECT *
FROM courses
WHERE cfu > 10;

-- COUNT
SELECT count(id)
FROM courses
WHERE cfu > 10;



-- 3. Tutti gli studenti che hanno più di 30 anni
SELECT *
FROM students
WHERE TIMESTAMPDIFF(YEAR, date_of_birth, CURDATE()) > 30; // TIMESTAMPDIFF calcola la differenza in anni //



-- 4. Corsi del primo semestre del primo anno
SELECT *
FROM courses
WHERE period = 'I semestre'
AND year = 1; // AND combina due condizioni //
